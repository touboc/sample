# path
%AppData%\yazi\config\yazi.toml

```c

[opener]
# Define Sakura Editor as an opener
# We use quotes and escape backslashes for the Windows path
sakura = [
    { run = '"C:\Program Files (x86)\sakura\sakura.exe" %s', block = false, desc = "Sakura Editor" }
]

[open]
rules = [
    # Match files with the .txt extension
    { name = "*.txt", use = "sakura" },
    
    # Optional: Match all text files if they don't have an extension 
    # but are identified as text by the system
    { mime = "text/*", use = "sakura" }
]
]
```
